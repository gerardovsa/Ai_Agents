pythonimport whois                       # Domain registration info

from datetime import datetime

def check_domain_age(domain):
    w = whois.whois(domain)
    creation_date = w.creation_date
    if isinstance(creation_date, list):
        creation_date = creation_date[0]
    
    age_days = (datetime.now() - creation_date).days
    
    if age_days < 365:
        return {"red_flag": True, "age_days": age_days}
    return {"red_flag": False, "age_days": age_days}

    async def find_credential_registry(self, profession, country):
        # Check cache first
        cache_key = f"{profession}_{country}"
        if cache_key in self.credential_registry:
            return self.credential_registry[cache_key]
        
        # Use LLM to find the registry
        prompt = f"""

        Provide:
        1. Official organization name
        2. Website URL
        3. Whether they have a public lookup tool
        4. If yes, the lookup URL format
        
        Be specific and factual.
        """

            model="claude-sonnet-4-20250514",

        )
        
        # Parse response and verify
        registry_info = self.parse_registry_info(response.content)
        
        # Verify the website exists
        verified = await self.verify_registry_website(registry_info['url'])
        
        if verified:
            # Cache it
            self.credential_registry[cache_key] = registry_info
            return registry_info
        
        return None

        self.docker_executor = await self.platform.spawn_docker_container(

        )

            "role": "user",
            "content": self.build_verification_prompt(candidate_data)
        }]
        
        result = await self.run_verification_loop(messages)
        
        # Step 3: Cleanup
        await self.platform.destroy_docker_container(self.docker_executor)
        
        return result
    
    async def run_verification_loop(self, messages):
        """
        The main loop where Claude uses computer tools
        """
        while True:
            # Call Claude with computer use tools enabled
            response = self.anthropic_client.messages.create(
                model="claude-sonnet-4-20250514",
                max_tokens=4096,
                tools=[{
                    "type": "computer_20241022",
                    "name": "computer",
                    "display_width_px": 1920,
                    "display_height_px": 1080,
                    "display_number": 1
                }, {
                    "type": "bash_20241022",
                    "name": "bash"
                }],
                messages=messages
            )
            
            # Handle tool use
            if response.stop_reason == "tool_use":
                tool_results = []
                
                for content_block in response.content:
                    if content_block.type == "tool_use":
                        # THIS IS WHERE YOUR PLATFORM EXECUTES THE TOOL
                        result = await self.execute_tool_in_docker(content_block)
                        
                        tool_results.append({
                            "type": "tool_result",
                            "tool_use_id": content_block.id,
                            "content": result
                        })
                
                # Continue the conversation
                messages.append({"role": "assistant", "content": response.content})
                messages.append({"role": "user", "content": tool_results})
                
            elif response.stop_reason == "end_turn":
                # Task complete!
                return self.extract_final_result(response)
    
    async def execute_tool_in_docker(self, tool_use_block):
        """
        Your platform executes Claude's tool request in the Docker container
        """

pythonclass MultiAgentVerificationPlatform:
    """
    Your platform running multiple verification agents in parallel
    """
    def __init__(self, max_concurrent_agents=5):
        self.platform = YourAIAgentPlatform()
        self.max_concurrent = max_concurrent_agents
        self.job_queue = asyncio.Queue()
        
    async def verify_batch(self, candidates):
        """
        Verify multiple candidates in parallel
        """
        # Add all jobs to queue
        for candidate in candidates:
            await self.job_queue.put(candidate)
        
        # Start worker agents
        workers = [
            asyncio.create_task(self.worker_agent(i))
            for i in range(self.max_concurrent)
        ]
        
        # Wait for all jobs to complete
        await self.job_queue.join()
        
        # Cancel workers
        for worker in workers:
            worker.cancel()
    
    async def worker_agent(self, worker_id):
        """
        Each worker processes jobs from the queue
        """
        while True:
            candidate = await self.job_queue.get()
            
            try:
                print(f"Worker {worker_id}: Starting verification for {candidate['name']}")
                result = await self.platform.run_verification_job(candidate)
                print(f"Worker {worker_id}: Completed - Risk score: {result['risk_score']}")
            except Exception as e:
                print(f"Worker {worker_id}: Error - {str(e)}")
            finally:
                self.job_queue.task_done()

pythonclass InteractiveSocialSearchAgent:
    """
    Agent that can request user input mid-process
    """
    def __init__(self):
        self.client = Anthropic()
        self.user_input_queue = asyncio.Queue()
        self.streaming_output = asyncio.Queue()
        
    async def search_with_interactive_auth(self, search_params):
        """
        Executes search and handles mid-process user input
        """
        messages = [{"role": "user", "content": self.build_search_prompt(search_params)}]
        
        async for event in self.execute_with_streaming(messages):
            # Emit updates to user interface
            await self.streaming_output.put(event)
            
            # Check if Claude needs user input
            if event['type'] == 'user_input_required':
                # Pause and wait for user
                user_response = await self.user_input_queue.get()
                
                # Continue conversation with user's response
                messages.append({
                    "role": "user", 
                    "content": user_response
                })
                
                # Resume execution
                continue
        
        return await self.streaming_output.get()
    
    async def execute_with_streaming(self, messages):
        """
        Stream responses and tool executions back to user
        """
        while True:
            # Start streaming response
            async with self.client.messages.stream(
                model="claude-sonnet-4-20250514",
                max_tokens=4096,
                tools=self.get_tools(),
                messages=messages
            ) as stream:
                
                # Stream text back to user in real-time
                async for text in stream.text_stream:
                    yield {
                        'type': 'text_delta',
                        'content': text
                    }
                
                # Get final message
                final_message = await stream.get_final_message()
                
                # Handle tool use
                if final_message.stop_reason == "tool_use":
                    tool_results = []
                    
                    for block in final_message.content:
                        if block.type == "tool_use":
                            # Execute tool and stream results
                            async for tool_event in self.execute_tool_with_streaming(block):
                                yield tool_event
                                
                                # Check if tool needs user input
                                if tool_event['type'] == 'needs_user_input':
                                    yield {
                                        'type': 'user_input_required',
                                        'prompt': tool_event['prompt'],
                                        'input_type': tool_event['input_type']  # 'text', '2fa_code', 'password', etc
                                    }
                                    
                                    # Wait for user input
                                    user_input = await self.user_input_queue.get()
                                    
                                    # Continue tool execution with user input
                                    tool_result = await self.continue_tool_with_input(
                                        block, 
                                        user_input
                                    )
                                else:
                                    tool_result = tool_event['result']
                            
                            tool_results.append({
                                "type": "tool_result",
                                "tool_use_id": block.id,
                                "content": tool_result
                            })
                    
                    # Continue conversation
                    messages.append({"role": "assistant", "content": final_message.content})
                    messages.append({"role": "user", "content": tool_results})
                    
                elif final_message.stop_reason == "end_turn":
                    yield {
                        'type': 'complete',
                        'final_message': final_message
                    }

                'event': 'user_input_required',
                'data': prompt_config
            })

            response = await self.websocket.receive_json()
            return response['user_input']
        else:
            # Fallback to queue-based (for non-WebSocket)
            await self.user_input_queue.put({
                'type': 'prompt',
                'data': prompt_config
            })
            
            response = await self.user_response_queue.get()
            return response

        """
        Stream updates to user interface
        """

from typing import Dict, Optional
from fastapi import WebSocket
import asyncio
import json
from datetime import datetime
import uuid

class WebSocketSession:
    """Manages a single WebSocket session with streaming and interaction"""
    
    def __init__(self, websocket: WebSocket, session_id: str):
        self.websocket = websocket
        self.session_id = session_id
        self.user_input_queue = asyncio.Queue()
        self.state = "active"
        self.metadata = {}
        self.created_at = datetime.now()
        
    async def send(self, message: dict):
        """Send message to client"""
        try:
            await self.websocket.send_json(message)
        except Exception as e:
            print(f"Error sending message: {e}")
            self.state = "disconnected"
    
    async def receive(self) -> dict:
        """Receive message from client"""
        try:
            data = await self.websocket.receive_json()
            return data
        except Exception as e:
            print(f"Error receiving message: {e}")
            self.state = "disconnected"
            return None
    
    async def request_user_input(self, prompt_config: dict) -> any:
        """Request input from user and wait for response"""
        request_id = str(uuid.uuid4())
        
        # Send request to client
        await self.send({
            'type': 'input_required',
            'request_id': request_id,
            **prompt_config
        })
        
        # Wait for response
        while True:
            response = await self.receive()
            if response and response.get('request_id') == request_id:
                return response.get('value')
    
    async def stream_text(self, text: str, source: str = "agent"):
        """Stream text content to client"""
        await self.send({
            'type': 'text_stream',
            'content': text,
            'source': source,
            'timestamp': datetime.now().isoformat()
        })
    
    async def stream_action(self, action: str, status: str = "running", metadata: dict = None):
        """Stream action/tool execution updates"""
        await self.send({
            'type': 'action',
            'action': action,
            'status': status,
            'metadata': metadata or {},
            'timestamp': datetime.now().isoformat()
        })
    
    async def stream_progress(self, current: int, total: int, message: str):
        """Stream progress updates"""
        await self.send({
            'type': 'progress',
            'current': current,
            'total': total,
            'message': message,
            'percentage': (current / total * 100) if total > 0 else 0
        })

class WebSocketManager:
    """Manages all active WebSocket sessions"""
    
    def __init__(self):
        self.active_sessions: Dict[str, WebSocketSession] = {}
    
    async def connect(self, websocket: WebSocket) -> WebSocketSession:
        """Create new session"""
        await websocket.accept()
        session_id = str(uuid.uuid4())
        session = WebSocketSession(websocket, session_id)
        self.active_sessions[session_id] = session
        
        await session.send({
            'type': 'connection_established',
            'session_id': session_id
        })
        
        return session
    
    def disconnect(self, session_id: str):
        """Remove session"""

@dataclass
class AgentStep:

class AgentExecutionState:
    """Maintains execution state for pause/resume"""
    
    def __init__(self, workflow_id: str):
        self.workflow_id = workflow_id
        self.current_step_index = 0
        self.step_results = {}
        self.context = {}
        self.paused = False
        self.completed = False
        self.error = None
        
    def to_dict(self):
        return {
            'workflow_id': self.workflow_id,
            'current_step': self.current_step_index,
            'step_results': self.step_results,
            'context': self.context,
            'paused': self.paused,
            'completed': self.completed,
            'error': str(self.error) if self.error else None
        }

class InteractiveAgentEngine:
    """
    Agent engine that can pause for user input and resume execution
    """
    
    def __init__(self, websocket_session: WebSocketSession):
        self.session = websocket_session
        self.anthropic = Anthropic()
        self.state = None
        self.workflow_steps: List[AgentStep] = []
        
    async def execute_workflow(
        self, 
        workflow_id: str,
        steps: List[AgentStep],
        initial_context: Dict = None
    ):
        """Execute a multi-step workflow with pause/resume capability"""
        
        # Initialize state
        self.state = AgentExecutionState(workflow_id)
        self.state.context = initial_context or {}
        self.workflow_steps = steps
        
        # Send workflow started
        await self.session.send({
            'type': 'workflow_started',
            'workflow_id': workflow_id,
            'total_steps': len(steps)
        })
        
        try:
            # Execute steps
            while self.state.current_step_index < len(steps):
                step = steps[self.state.current_step_index]
                
                # Execute step
                result = await self.execute_step(step)
                
                # Store result
                self.state.step_results[step.name] = result
                
                # Move to next step
                self.state.current_step_index += 1
                
                # Update progress
                await self.session.stream_progress(
                    self.state.current_step_index,
                    len(steps),
                    f"Completed: {step.name}"
                )
            
            # Mark as completed
            self.state.completed = True
            
            await self.session.send({
                'type': 'workflow_completed',
                'workflow_id': workflow_id,
                'results': self.state.step_results
            })
            
            return self.state.step_results
            
        except Exception as e:
            self.state.error = e
            await self.session.send({
                'type': 'workflow_error',
                'workflow_id': workflow_id,
                'error': str(e)
            })
            raise
    
    async def execute_step(self, step: AgentStep) -> Any:
        """Execute a single step"""
        
        # Stream step start
        await self.session.stream_action(
            step.name,
            status="starting",
            metadata={'description': step.description}
        )
        
        # Check if step requires user input
        if step.requires_input and step.input_config:
            user_input = await self.session.request_user_input(step.input_config)
            self.state.context[f'{step.name}_user_input'] = user_input
        
        # Execute the step function
        try:
            result = await step.function(
                context=self.state.context,
                session=self.session
            )
            
            # Stream step completion
            await self.session.stream_action(
                step.name,
                status="completed",
                metadata={'result': result}
            )
            
            return result
            
        except Exception as e:
            await self.session.stream_action(
                step.name,
                status="failed",
                metadata={'error': str(e)}
            )
            raise
    
    async def execute_with_claude(
        self,
        prompt: str,
        tools: List = None,
        system: str = None
    ) -> Dict:
        """
        Execute Claude API call with streaming and interactive tool use
        """
        messages = [{"role": "user", "content": prompt}]
        tool_results_accumulated = []
        
        while True:
            # Stream thinking
            await self.session.stream_text("Claude is thinking...", source="system")
            
            # Make API call with streaming
            async with self.anthropic.messages.stream(
                model="claude-sonnet-4-20250514",
                max_tokens=4096,
                messages=messages,
                tools=tools or [],
                system=system
            ) as stream:
                
                # Stream text as it comes
                full_text = ""
                async for text in stream.text_stream:
                    await self.session.stream_text(text, source="claude")
                    full_text += text
                
                # Get final message
                final_message = await stream.get_final_message()
            
            # Handle tool use
            if final_message.stop_reason == "tool_use":
                tool_results = []
                
                for block in final_message.content:
                    if block.type == "tool_use":
                        # Stream tool execution
                        await self.session.stream_action(
                            f"Executing: {block.name}",
                            status="running",
                            metadata={'input': block.input}
                        )
                        
                        # Execute tool with interactive capability

                        )

                            "type": "tool_result",
                            "tool_use_id": block.id,

                        })
                        
                        tool_results_accumulated.append({
                            'tool': block.name,
                            'input': block.input,
                            'result': tool_result
                        })
                
                # Continue conversation
                messages.append({"role": "assistant", "content": final_message.content})
                messages.append({"role": "user", "content": tool_results})
                
            elif final_message.stop_reason == "end_turn":
                # Conversation complete
                return {
                    'final_text': full_text,
                    'tool_results': tool_results_accumulated,
                    'full_response': final_message
                }

import { useState, useEffect, useRef, useCallback } from 'react';

export function useAgentWebSocket(url) {
    const [messages, setMessages] = useState([]);
    const [connected, setConnected] = useState(false);
    const [sessionId, setSessionId] = useState(null);
    const [currentStep, setCurrentStep] = useState(null);
    const [progress, setProgress] = useState({ current: 0, total: 0, percentage: 0 });
    const [inputRequired, setInputRequired] = useState(null);
    
    const ws = useRef(null);
    const messageHandlers = useRef({});
    
    useEffect(() => {
        // Connect WebSocket
        ws.current = new WebSocket(url);
        
        ws.current.onopen = () => {
            console.log('WebSocket connected');
            setConnected(true);
        };
        
        ws.current.onmessage = (event) => {
            const message = JSON.parse(event.data);
            console.log('Received:', message);
            
            // Add to messages
            setMessages(prev => [...prev, message]);
            
            // Handle specific message types
            handleMessage(message);
        };
        
        ws.current.onclose = () => {
            console.log('WebSocket disconnected');
            setConnected(false);
        };
        
        ws.current.onerror = (error) => {
            console.error('WebSocket error:', error);
        };
        
        return () => {
            if (ws.current) {
                ws.current.close();
            }
        };
    }, [url]);
    
    const handleMessage = (message) => {
        switch (message.type) {
            case 'connection_established':
                setSessionId(message.session_id);
                break;
                
            case 'workflow_started':
                setProgress({ current: 0, total: message.total_steps, percentage: 0 });
                break;
                
            case 'action':
                setCurrentStep(message.action);
                break;
                
            case 'progress':
                setProgress({
                    current: message.current,
                    total: message.total,
                    percentage: message.percentage
                });
                break;
                
            case 'input_required':
                setInputRequired(message);
                break;
                
            case 'workflow_completed':
                setCurrentStep(null);
                setProgress({ current: message.total_steps, total: message.total_steps, percentage: 100 });
                break;
        }
        
        // Call registered handlers
        if (messageHandlers.current[message.type]) {
            messageHandlers.current[message.type](message);
        }
    };
    
    const send = useCallback((message) => {
        if (ws.current && connected) {
            ws.current.send(JSON.stringify(message));
        }
    }, [connected]);
    
    const startWorkflow = useCallback((workflowType, data) => {
        send({
            type: 'start_workflow',
            workflow_type: workflowType,
            data: data
        });
    }, [send]);
    
    const respondToInput = useCallback((requestId, value) => {
        send({
            type: 'user_response',
            request_id: requestId,
            value: value
        });
        setInputRequired(null);
    }, [send]);
    
    const registerHandler = useCallback((messageType, handler) => {
        messageHandlers.current[messageType] = handler;
    }, []);
    
    return {
        connected,
        sessionId,
        messages,
        currentStep,
        progress,
        inputRequired,
        startWorkflow,
        respondToInput,
        registerHandler,
        send
    };
}

                    </div>
                    <button
                        onClick={handleStart}
                        disabled={!connected || !candidateName}
                        className="start-button"
                    >
                        Start Workflow
                    </button>
                </div>
            )}
            
            {running && (
                <>
                    <ProgressBar progress={progress} currentStep={currentStep} />
                    
                    <div className="messages-container">
                        <MessageList messages={messages} />
                        <div ref={messagesEndRef} />
                    </div>
                    
                    {inputRequired && (
                        <InputModal
                            inputConfig={inputRequired}
                            onSubmit={(value) => respondToInput(inputRequired.request_id, value)}
                            onCancel={() => respondToInput(inputRequired.request_id, null)}
                        />
                    )}
                </>
            )}
        </div>
    );
}

function ProgressBar({ progress, currentStep }) {
    return (
        <div className="progress-section">
            <div className="progress-header">
                <span>Progress: {progress.current} / {progress.total}</span>
                {currentStep && <span className="current-step">{currentStep}</span>}
            </div>
            <div className="progress-bar">
                <div
                    className="progress-fill"
                    style={{ width: `${progress.percentage}%` }}
                />
            </div>
        </div>
    );
}

function MessageList({ messages }) {
    return (
        <div className="messages-list">
            {messages.map((msg, idx) => (
                <Message key={idx} message={msg} />
            ))}
        </div>
    );
}

function Message({ message }) {
    const getMessageClass = () => {
        switch (message.type) {
            case 'text_stream':
                return `message message-${message.source}`;
            case 'action':
                return `message message-action status-${message.status}`;
            case 'progress':
                return 'message message-progress';
            case 'workflow_completed':
                return 'message message-success';
            case 'error':
                return 'message message-error';
            default:
                return 'message message-system';
        }
    };
    
    const renderContent = () => {
        switch (message.type) {
            case 'text_stream':
                return (
                    <div className="message-content">
                        <span className="source-badge">{message.source}</span>
                        <pre className="text-content">{message.content}</pre>
                    </div>
                );
                
            case 'action':
                return (
                    <div className="message-content">
                        <span className="action-icon">⚡</span>
                        <span className="action-name">{message.action}</span>
                        <span className={`status-badge status-${message.status}`}>
                            {message.status}
                        </span>
                        {message.metadata && (
                            <div className="metadata">
                                {JSON.stringify(message.metadata, null, 2)}
                            </div>
                        )}
                    </div>
                );
                
            case 'workflow_started':
                return (
                    <div className="message-content">
                        <span className="icon">🚀</span>
                        <span>Workflow started: {message.workflow_id}</span>
                        <span className="total-steps">{message.total_steps} steps</span>
                    </div>
                );
                
            case 'workflow_completed':
                return (
                    <div className="message-content">
                        <span className="icon">✅</span>
                        <span>Workflow completed successfully!</span>
                        <details className="results">
                            <summary>View Results</summary>
                            <pre>{JSON.stringify(message.results, null, 2)}</pre>
                        </details>
                    </div>
                );
                
            case 'error':
                return (
                    <div className="message-content">
                        <span className="icon">❌</span>
                        <span>Error: {message.message}</span>
                    </div>
                );
                
            default:
                return (
                    <div className="message-content">
                        <pre>{JSON.stringify(message, null, 2)}</pre>
                    </div>
                );
        }
    };
    
    return (
        <div className={getMessageClass()}>
            {renderContent()}
            {message.timestamp && (
                <span className="timestamp">
                    {new Date(message.timestamp).toLocaleTimeString()}
                </span>
            )}
        </div>
    );
}

function InputModal({ inputConfig, onSubmit, onCancel }) {
    const [value, setValue] = useState('');
    
    const handleSubmit = (e) => {
        e.preventDefault();
        onSubmit(value);
    };
    
    const renderInput = () => {
        switch (inputConfig.input_type) {
            case '2fa_code':
                return (
                    <input
                        type="text"
                        value={value}
                        onChange={(e) => setValue(e.target.value)}
                        placeholder={inputConfig.placeholder || '123456'}
                        maxLength="6"
                        autoFocus
                        className="input-field code-input"
                    />
                );
                
            case 'choice':
                return (
                    <div className="choice-options">
                        {inputConfig.options?.map((option, idx) => (
                            <button
                                key={idx}
                                onClick={() => {
                                    setValue(option.id);
                                    onSubmit(option.id);
                                }}
                                className="choice-button"
                            >
                                {option.label}
                                {option.description && (
                                    <span className="choice-description">{option.description}</span>
                                )}
                            </button>
                        ))}
                    </div>
                );
                
            case 'confirmation':
                return (
                    <div className="confirmation-buttons">
                        <button onClick={() => onSubmit(true)} className="confirm-yes">
                            Yes, Continue
                        </button>
                        <button onClick={() => onSubmit(false)} className="confirm-no">
                            No, Skip
                        </button>
                    </div>
                );
                
            default:
                return (
                    <input
                        type="text"
                        value={value}
                        onChange={(e) => setValue(e.target.value)}
                        placeholder={inputConfig.placeholder}
                        autoFocus
                        className="input-field"
                    />
                );
        }
    };
    
    return (
        <div className="modal-overlay">
            <div className="modal">
                <div className="modal-header">
                    <h3>Input Required</h3>
                </div>
                <div className="modal-body">
                    <p className="prompt">{inputConfig.prompt}</p>
                    {inputConfig.message && (
                        <p className="message">{inputConfig.message}</p>
                    )}
                    <form onSubmit={handleSubmit}>
                        {renderInput()}
                        {inputConfig.input_type !== 'choice' && inputConfig.input_type !== 'confirmation' && (
                            <div className="button-group">
                                <button type="submit" className="submit-button">
                                    Submit
                                </button>
                                <button type="button" onClick={onCancel} className="cancel-button">
                                    Cancel
                                </button>
                            </div>
                        )}
                    </form>
                </div>
            </div>
        </div>
    );
}
