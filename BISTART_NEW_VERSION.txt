function BISTART {
    $ai_agents_path = 'C:\Users\gpoli\GIT\AI_agents\AI_infrastructure'
    $flask_script = 'flask_app.py'
    $ui_html = 'C:\Users\gpoli\GIT\AI_agents\UI\business-ai-platform-v2.html'

    if (Test-Path (Join-Path $ai_agents_path $flask_script)) {
        Write-Host '' -ForegroundColor Green
        Write-Host '===================================================================' -ForegroundColor Cyan
        Write-Host '  Launching AI Agents Business Intelligence Platform' -ForegroundColor Yellow
        Write-Host '===================================================================' -ForegroundColor Cyan
        
        # Step 1: Kill any existing Flask processes on port 5001
        Write-Host '  [1/3] Checking for existing Flask processes on port 5001...' -ForegroundColor Yellow
        $existingProcess = Get-NetTCPConnection -LocalPort 5001 -ErrorAction SilentlyContinue | 
                          Select-Object -ExpandProperty OwningProcess -First 1
        
        if ($existingProcess) {
            Write-Host "        Stopping existing Flask process (PID: $existingProcess)..." -ForegroundColor Yellow
            Stop-Process -Id $existingProcess -Force -ErrorAction SilentlyContinue
            Start-Sleep -Seconds 2
            Write-Host '        [OK] Existing Flask stopped' -ForegroundColor Green
        } else {
            Write-Host '        [OK] No existing Flask process found' -ForegroundColor Green
        }
        
        Write-Host ''
        Write-Host '  Flask AI:  ' -NoNewline -ForegroundColor White
        Write-Host 'http://localhost:5001' -ForegroundColor Green
        Write-Host '  Health:    ' -NoNewline -ForegroundColor White
        Write-Host 'http://localhost:5001/health' -ForegroundColor Green
        Write-Host '===================================================================' -ForegroundColor Cyan
        Write-Host ''

        # Step 2: Start Flask in separate window
        Write-Host '  [2/3] Starting Flask server...' -ForegroundColor Cyan
        Set-Location $ai_agents_path
        
        # Start Flask in a NEW PowerShell window
        $flaskProcess = Start-Process powershell -ArgumentList "-NoExit", "-Command", "python flask_app.py" -WorkingDirectory $ai_agents_path -PassThru
        Write-Host "        Flask PID: $($flaskProcess.Id)" -ForegroundColor Gray
        Write-Host "        Flask window opened in separate terminal" -ForegroundColor Gray
        Write-Host ''
        Write-Host "        Waiting 15 seconds for Flask to fully initialize..." -ForegroundColor Yellow
        Write-Host "        (Flask starts + auto-reloader restarts = ~15 sec)" -ForegroundColor Gray
        
        # Progress bar
        for ($i = 1; $i -le 15; $i++) {
            Start-Sleep -Seconds 1
            Write-Host '.' -NoNewline -ForegroundColor Yellow
        }
        
        Write-Host ''
        Write-Host '        [OK] Flask should be ready!' -ForegroundColor Green

        # Step 3: Open UI in browser
        Write-Host '  [3/3] Opening UI in browser...' -ForegroundColor Cyan
        Start-Sleep -Seconds 1
        
        if (Test-Path $ui_html) {
            Write-Host "        UI Path: $ui_html" -ForegroundColor Gray
            Start-Process $ui_html
            Start-Sleep -Seconds 2
            Write-Host '        [OK] Browser opened' -ForegroundColor Green
            Write-Host ''
            Write-Host '===================================================================' -ForegroundColor Cyan
            Write-Host '  [READY] Flask running (PID: ' -NoNewline -ForegroundColor Green
            Write-Host "$($flaskProcess.Id)" -NoNewline -ForegroundColor White
            Write-Host ') + UI opened' -ForegroundColor Green
            Write-Host '===================================================================' -ForegroundColor Cyan
            Write-Host ''
            Write-Host '  To stop Flask:' -ForegroundColor Gray
            Write-Host "    Stop-Process -Id $($flaskProcess.Id)" -ForegroundColor Gray
            Write-Host '  Or press Ctrl+C in the Flask window' -ForegroundColor Gray
            Write-Host ''
            Write-Host '  NOTE: If UI shows "Backend disconnected", wait a few more seconds' -ForegroundColor Yellow
            Write-Host '        and refresh the browser page. Flask may still be initializing.' -ForegroundColor Yellow
            Write-Host ''
        } else {
            Write-Host '        [ERROR] UI file not found at:' -ForegroundColor Red
            Write-Host "        $ui_html" -ForegroundColor Gray
            Write-Host ''
            Write-Host '  Flask is running but UI could not be opened.' -ForegroundColor Yellow
            Write-Host "  You can manually open: $ui_html" -ForegroundColor Yellow
        }
    } else {
        Write-Host '' -ForegroundColor Red
        Write-Host ' [ERROR] AI Agents Flask script not found at:' -ForegroundColor Red
        Write-Host (Join-Path $ai_agents_path $flask_script) -ForegroundColor Yellow
        Write-Host ''
        Write-Host 'Expected location: C:\Users\gpoli\GIT\AI_agents\AI_infrastructure\flask_app.py' -ForegroundColor Gray
        Write-Host ''
    }
}
